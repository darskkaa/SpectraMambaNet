def run(cmd, log_path=None, watch=None, log_mode="w"):
    """Stream a child process live; optionally tee to log_path and assert on `watch`."""
    # GITHUB_TOKEN must be *in the env*: setup_repo() falls back to userdata.get(), but
    # that has no secrets channel in a subprocess (colab_train_launcher.py:255-266), so
    # without this the clone of the private repo fails the same way the raw fetch did.
    #
    # WACV_RUN is what suffixes CKPT_BACKUP (colab_train_launcher.py:51-53). Unset, every
    # run shares WACV_2027_Checkpoints and the copytree(dirs_exist_ok=True) overwrites the
    # previous run's last.ckpt in place -- the launcher's own comment records that this
    # destroyed a baseline once. It also breaks the resume path in this cell: FRESH is
    # computed from WACV_2027_Checkpoints_fin1, which would never appear, so every relaunch
    # after a crash would pass --fresh and wipe the run.
    env = dict(os.environ, PYTHONUNBUFFERED="1", GITHUB_TOKEN=TOKEN, WACV_RUN=RUN)
    proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                            stdin=subprocess.DEVNULL, text=True, bufsize=1, env=env)
    buf = []
    sink = open(log_path, log_mode, buffering=1, encoding="utf-8") if log_path else None
    checked = False
    try:
        for line in proc.stdout:
            sys.stdout.write(line)
            if sink:
                sink.write(line)
            buf.append(line)
            if watch and not checked and line.startswith("[run-config]"):
                checked = True
                missing = [t for t in watch if t not in line]
                if missing:
                    proc.kill()
                    print("\n" + "=" * 70)
                    print("ABORTED — [run-config] does not match the intended fin1 config.")
                    print("Missing:", ", ".join(missing))
                    print("=" * 70)
                    break
                print("\n[check] [run-config] matches all "
                      f"{len(watch)} required settings.\n")
    finally:
        proc.stdout.close()
        proc.wait()
        if sink:
            sink.close()
    return proc.returncode, "".join(buf)


def die(msg):
    print("\n" + "=" * 70)
    print("STOP —", msg)
    print("=" * 70)
    raise SystemExit(1)


def read_token():
    """Colab secret -> env var. Read here, in the kernel, because only the kernel has
    the secrets channel."""
    tok = os.environ.get("GITHUB_TOKEN")
    if tok:
        return tok.strip()
    try:
        from google.colab import userdata
        return (userdata.get("GITHUB_TOKEN") or "").strip()
    except Exception as e:
        print(f"[setup] userdata.get('GITHUB_TOKEN') failed: {type(e).__name__}")
        return ""


def fetch_launcher(dest):
    """Pull colab_train_launcher.py off the private repo.

    raw.githubusercontent.com answers a private path with 404 rather than 403, which is
    why the unauthenticated urlretrieve looked like a missing file. The contents API with
    a token is the supported way in; Accept: v3.raw returns file bytes, not JSON.
    """
    url = f"https://api.github.com/repos/{REPO}/contents/colab_train_launcher.py?ref={BRANCH}"
    req = urllib.request.Request(url, headers={
        "Authorization": f"Bearer {TOKEN}",
        "Accept": "application/vnd.github.v3.raw",
        "X-GitHub-Api-Version": "2022-11-28",
    })
    try:
        with urllib.request.urlopen(req, timeout=60) as r:
            body = r.read()
    except urllib.error.HTTPError as e:
        if e.code in (401, 403):
            die(f"GitHub rejected GITHUB_TOKEN ({e.code}). The token is expired or lacks "
                f"read access to {REPO}. A fine-grained PAT needs Contents: Read on that repo.")
        if e.code == 404:
            die(f"404 from the contents API for {REPO}@{BRANCH}. Either GITHUB_TOKEN cannot "
                f"see this private repo, or the branch name is wrong.")
        raise
    # A proxy/login page written to disk would fail later as a Python SyntaxError, 20
    # minutes into the pre-flight. Check for something only the real launcher contains.
    if b"EXPECTED_TRAIN_IMAGES" not in body:
        die(f"fetched {len(body)} bytes that are not colab_train_launcher.py.")
    with open(dest, "wb") as f:
        f.write(body)
    return len(body)
